% 1. PyObject Internal Struct Anatomy
\newcommand{\DrawPyObjectStruct}[4]{%
  % #1: Address (e.g. 0x100A)
  % #2: RefCount (e.g. 1)
  % #3: Type (e.g. <class 'int'>)
  % #4: Payload (e.g. 42)
  \begin{tikzpicture}[font=\ttfamily\scriptsize, every node/.style={align=center}]
    % Object Boundary Container
    \draw[draw=UpGradRule,fill=UpGradMist,rounded corners=2mm,thick] (0,0) rectangle (6.8,2.4);
    \node[anchor=north west,font=\tiny\bfseries\color{UpGradSlate}] at (0.2,2.25) {Heap PyObject at #1};
    
    % Slot 1: ob_refcnt
    \draw[draw=AWSBlue,fill=AWSBlue!12,rounded corners=1mm,thick] (0.3,0.3) rectangle (2.1,1.8);
    \node[font=\tiny\color{AWSBlue}] at (1.2,1.5) {ob\_refcnt};
    \node[font=\small\bfseries\color{UpGradNight}] at (1.2,0.9) {#2};
    \node[font=\tiny\color{UpGradSlate}] at (1.2,0.5) {8 Bytes};
    
    % Slot 2: ob_type*
    \draw[draw=AWSOrange,fill=AWSOrange!12,rounded corners=1mm,thick] (2.4,0.3) rectangle (4.4,1.8);
    \node[font=\tiny\color{AWSOrange}] at (3.4,1.5) {ob\_type*};
    \node[font=\scriptsize\bfseries\color{UpGradNight}] at (3.4,0.9) {#3};
    \node[font=\tiny\color{UpGradSlate}] at (3.4,0.5) {8 Bytes};
    
    % Slot 3: ob_ival / payload
    \draw[draw=AWSEmerald,fill=AWSEmerald!12,rounded corners=1mm,thick] (4.7,0.3) rectangle (6.5,1.8);

% 2. Pointer Rebinding Flipbook Diagram
\newcommand{\DrawPointerRebinding}[1]{%
  % #1: state = 1 (x=42), 2 (x="hello")
  \begin{tikzpicture}[font=\ttfamily\scriptsize, scale=0.92, transform shape]
    % Stack Frame Area
    \draw[draw=UpGradSlate!40,fill=UpGradCode,rounded corners=1.5mm] (0,0) rectangle (2.8,3.2);
    \node[anchor=north west,font=\tiny\bfseries\color{UpGradSlate}] at (0.2,3.0) {STACK (Names)};
    
    \node[draw=UpGradNight,fill=white,rounded corners=1mm,minimum width=2.2cm,minimum height=0.7cm,thick] (stackx) at (1.4,1.8) {
      \textbf{\color{UpGradRed}x:\enspace}\ifnum#1=1 \color{AWSBlue}0x100A \else \color{AWSBlue}0x200B \fi
    };
    
    % Heap Objects Area
    \draw[draw=UpGradSlate!40,fill=UpGradMist,rounded corners=1.5mm] (4.2,0) rectangle (10.2,3.2);
    \node[anchor=north west,font=\tiny\bfseries\color{UpGradSlate}] at (4.4,3.0) {HEAP (Allocated PyObjects)};
    
    % Object 1: 42 at 0x100A
    \ifnum#1=1
      \node[draw=AWSEmerald,fill=white,rounded corners=1.5mm,thick,minimum width=5.2cm,minimum height=1.0cm] (obj1) at (7.2,2.2) {
        \textbf{0x100A}\enspace{\color{UpGradRule}\textbar}\enspace\textbf{int: 42}\enspace{\color{UpGradSlate}(refcnt=1)}
      };
      \draw[upgrad/pointer,AWSBlue,very thick] (stackx.east) -- (obj1.west);
    \else
      \node[draw=UpGradRed!70,dashed,fill=white,rounded corners=1.5mm,thick,minimum width=5.2cm,minimum height=1.0cm] (obj1) at (7.2,2.2) {
        \textbf{\color{UpGradSlate}0x100A}\enspace{\color{UpGradRule}\textbar}\enspace{\color{UpGradSlate}int: 42}\enspace{\color{UpGradRed}(refcnt=0 $\to$ GC)}
      };
      \node[draw=AWSEmerald,fill=white,rounded corners=1.5mm,thick,minimum width=5.2cm,minimum height=1.0cm] (obj2) at (7.2,0.8) {
        \textbf{0x200B}\enspace{\color{UpGradRule}\textbar}\enspace\textbf{str: "hello"}\enspace{\color{UpGradSlate}(refcnt=1)}
      };
      \draw[upgrad/pointer,AWSBlue,very thick] (stackx.east) -- (obj2.west);
      \draw[dashed,UpGradRed,thick] (stackx.east) -- (obj1.west);
    \fi
  \end{tikzpicture}%
}

% 3. Copy Semantics Reference Graph Diagram
\newcommand{\DrawCopySemanticsDiagram}[1]{%
  % #1: mode = 1 (Assignment/Alias), 2 (Shallow Copy), 3 (Deep Copy)
  \begin{tikzpicture}[font=\ttfamily\scriptsize, scale=0.88, transform shape]
    % Stack Names
    \node[draw=UpGradNight,fill=white,rounded corners=1mm,minimum width=1.8cm,minimum height=0.6cm,thick] (na) at (0,2.2) {\textbf{\color{UpGradRed}a}};
    \node[draw=UpGradNight,fill=white,rounded corners=1mm,minimum width=1.8cm,minimum height=0.6cm,thick] (nb) at (0,0.6) {\textbf{\color{AWSBlue}b}};
    
    % Outer Container A
    \node[draw=UpGradRed,fill=UpGradMist,rounded corners=1.5mm,thick,minimum width=3.4cm,minimum height=0.9cm] (outA) at (3.8,2.2) {
      \textbf{List A: [1, \color{UpGradRed}0x200\color{UpGradInk}]}
    };
    \draw[upgrad/pointer,UpGradRed,thick] (na.east) -- (outA.west);
    
    % Shared / Duplicated Inner Object
    \ifnum#1=1
      % Assignment
      \draw[upgrad/pointer,AWSBlue,thick] (nb.east) -- (outA.west);
      \node[draw=AWSEmerald,fill=white,rounded corners=1.5mm,thick,minimum width=2.4cm,minimum height=0.8cm] (inner) at (8.2,1.4) {
        \textbf{0x200: [2, 3]}
      };
      \draw[upgrad/pointer,UpGradRed,thick] (outA.east) -- (inner.west);
      \node[anchor=north west,font=\tiny\bfseries\color{UpGradRedDark}] at (2.4,0.0) {Alias: Both names point to identical outer container (0x100)};
    \fi
    
    \ifnum#1=2
      % Shallow Copy
      \node[draw=AWSBlue,fill=UpGradMist,rounded corners=1.5mm,thick,minimum width=3.4cm,minimum height=0.9cm] (outB) at (3.8,0.6) {
        \textbf{List B: [1, \color{UpGradRed}0x200\color{UpGradInk}]}
      };
      \draw[upgrad/pointer,AWSBlue,thick] (nb.east) -- (outB.west);
      
      \node[draw=AWSEmerald,fill=white,rounded corners=1.5mm,thick,minimum width=2.4cm,minimum height=0.8cm] (inner) at (8.2,1.4) {
        \textbf{0x200: [2, 3]}
      };
      \draw[upgrad/pointer,UpGradRed,thick] (outA.east) -- (inner.west);
      \draw[upgrad/pointer,AWSBlue,thick] (outB.east) -- (inner.west);
      \node[anchor=north west,font=\tiny\bfseries\color{AWSOrange}] at (2.0, -0.2) {Shallow Copy: Distinct outer containers (0x100 vs 0x300), SHARED inner list (0x200)!};
    \fi
    
    \ifnum#1=3
      % Deep Copy
      \node[draw=AWSBlue,fill=UpGradMist,rounded corners=1.5mm,thick,minimum width=3.4cm,minimum height=0.9cm] (outB) at (3.8,0.6) {
        \textbf{List B: [1, \color{AWSBlue}0x400\color{UpGradInk}]}
      };
      \draw[upgrad/pointer,AWSBlue,thick] (nb.east) -- (outB.west);
      
      \node[draw=AWSEmerald,fill=white,rounded corners=1.5mm,thick,minimum width=2.4cm,minimum height=0.7cm] (innerA) at (8.2,2.2) {
        \textbf{0x200: [2, 3]}
      };
      \node[draw=AWSEmerald,fill=white,rounded corners=1.5mm,thick,minimum width=2.4cm,minimum height=0.7cm] (innerB) at (8.2,0.6) {
        \textbf{0x400: [2, 3]}
      };
      \draw[upgrad/pointer,UpGradRed,thick] (outA.east) -- (innerA.west);
      \draw[upgrad/pointer,AWSBlue,thick] (outB.east) -- (innerB.west);
      \node[anchor=north west,font=\tiny\bfseries\color{AWSEmerald}] at (2.0, -0.2) {Deep Copy: Complete recursive cloning. Zero shared references between A and B!};
    \fi
  \end{tikzpicture}%
}
