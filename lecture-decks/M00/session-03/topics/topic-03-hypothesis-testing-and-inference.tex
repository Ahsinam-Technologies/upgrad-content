\begin{frame}{Cognitive Objectives}
  \begin{LearningObjectives}{By the end of this topic, learners will be able to:}
    \begin{itemize}\setlength{\itemsep}{0.28em}
      \item \textbf{Formulate testable hypotheses} with the Null Hypothesis ($H_0$) as the conservative presumption of zero effect.
      \item \textbf{Interpret $p$-values accurately} as the probability of observing data at least as extreme under $H_0$ (not the probability that $H_0$ is true).
      \item \textbf{Balance Type I ($\alpha$) vs Type II ($\beta$) errors} to maximize statistical power ($1-\beta$) in production A/B tests.
      \item \textbf{Execute non-parametric permutation tests} in NumPy to evaluate metrics without asymptotic normality assumptions.
    \end{itemize}
  \end{LearningObjectives}
\end{frame}

\begin{frame}[fragile]{The A/B Testing Dilemma}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{UpGradRed}The Business Question: Real or Luck?}\par\vspace{0.08cm}
      \begin{itemize}\setlength{\itemsep}{0.20em}\small
        \item Control ($A$): 2,000 users $\to$ 2.1\% conversion.
        \item Treatment ($B$): 2,000 users $\to$ 2.5\% conversion (+0.4\% lift).
        \item Product Manager wants to ship immediately!
        \item \textcolor{UpGradRed}{\textbf{Socratic Probe:}} Is this +0.4\% lift a real behavioral change or just random coin-flip variance?
      \end{itemize}
    \end{column}
    \hfill{\color{UpGradRule}\vrule width .5pt}\hfill
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{AWSEmerald}The "Aha!" Discovery: Proof by Contradiction}\par\vspace{0.08cm}
      \begin{itemize}\setlength{\itemsep}{0.18em}\small
        \item \textbf{Null Hypothesis ($H_0$):} Treatment has \textbf{zero effect} ($\mu_B - \mu_A = 0$). Presumption of innocence!
        \item \textbf{Simulate the Null Universe:} What is the distribution of differences if Variant B is completely useless?
        \item \textbf{$p$-Value:} How rarely would random noise generate a lift $\ge +0.4\%$?
      \end{itemize}
      \vspace{0.04cm}
      \TakeawayBanner{$p$-value measures surprise under the null hypothesis.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Production Checkpoint: Non-Parametric Permutation Testing}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{UpGradNight}Permutation Test Algorithm}\par\vspace{0.06cm}
      \begin{enumerate}\setlength{\itemsep}{0.14em}\small
        \item Compute observed metric delta $\Delta^* = \bar{B} - \bar{A}$.
        \item Pool all observations ($N_A + N_B$).
        \item Reshuffle group labels $K = 10,000$ times.
        \item Fraction of shuffled trials with $\Delta_{\text{null}} \ge \Delta^*$ gives the exact empirical $p$-value.
      \end{enumerate}
      \vspace{0.04cm}
      \TakeawayBanner{Permutation tests eliminate all parametric assumptions.}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{UpGradRed}NumPy Implementation}\par\vspace{0.04cm}
\begin{CodeBlock}{Python}
import numpy as np

def permutation_test(ctrl, trt, k=10_000):
    obs_diff = np.mean(trt) - np.mean(ctrl)
    pooled = np.concatenate([ctrl, trt])
    n = len(ctrl)
    null_diffs = np.empty(k)
    for i in range(k):
        np.random.shuffle(pooled)
        null_diffs[i] = np.mean(pooled[n:]) - np.mean(pooled[:n])
    return float(np.mean(null_diffs >= obs_diff))
\end{CodeBlock}
    \end{column}
  \end{columns}
\end{frame}
