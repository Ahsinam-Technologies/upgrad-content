\begin{frame}[fragile]{The Dependency Collision Catastrophe}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{UpGradRed}The Collision: Global Package Overwrites}\par\vspace{0.08cm}
      \begin{itemize}\setlength{\itemsep}{0.20em}\small
        \item \textbf{Legacy Model API (Project A):} Requires \InlineCode{pandas==1.5.3} and \InlineCode{numpy<1.24}.
        \item \textbf{New GenAI Pipeline (Project B):} Requires \InlineCode{pandas>=2.2.0} and \InlineCode{numpy>=1.26}.
        \item Running \InlineCode{pip install pandas==2.2} globally silently overwrites the files in \InlineCode{/usr/lib/python3.11/site-packages/}.
        \item \textcolor{UpGradRed}{\textbf{Production Outage:}} Project A's production inference server crashes with \InlineCode{AttributeError}!
      \end{itemize}
    \end{column}
    \hfill{\color{UpGradRule}\vrule width .5pt}\hfill
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{AWSEmerald}The "Aha!" Discovery: Path Sandboxing}\par\vspace{0.08cm}
      \begin{itemize}\setlength{\itemsep}{0.18em}\small
        \item A Virtual Environment is not a VM or container; it is simply a \textbf{directory tree with a redirected \InlineCode{sys.prefix}}!
        \item CPython checks \InlineCode{pyvenv.cfg} on startup.
        \item \InlineCode{sys.path} prepends the local \InlineCode{.venv/lib/python/site-packages/} ahead of global directories.
        \item Multiple isolated environments coexist on the same machine without mutual interference.
      \end{itemize}
      \vspace{0.04cm}
      \TakeawayBanner{Never execute \InlineCode{pip install} outside an active virtual environment.}
    \end{column}
  \end{columns}
\end{frame}

        \item Global hardlink caching avoids copying duplicate wheels across projects.
        \item Replaces \InlineCode{pip}, \InlineCode{pip-tools}, \InlineCode{virtualenv}, \InlineCode{poetry}, and \InlineCode{pyenv}.
      \end{itemize}
