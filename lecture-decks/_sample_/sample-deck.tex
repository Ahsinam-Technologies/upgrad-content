\begin{frame}{A normal content slide: define the learning path}

      Which subscribers should receive a retention offer this week, given a limited outreach budget?
    \end{ConceptBox}
    \vspace{0.25cm}

      \item Decision owner: customer-success lead
      \item Unit: one active subscriber
      \item Outcome horizon: churn in the next 30 days
    \end{itemize}

      Recent support friction, low tenure and a falling engagement trend may raise churn risk.
    \end{ExampleBox}
    \vspace{0.25cm}

      \item Train on observations available before outreach.
      \item Compare against a no-model targeting baseline.
      \item Review false positives with the operations team.
    \end{itemize}
  }

\begin{frame}{Dense teaching slide: the minimum viable data contract}
  \small
  \begin{UpGradTable}{>{\raggedright\arraybackslash}p{.23\linewidth} >{\raggedright\arraybackslash}p{.23\linewidth} X}
    \TableHeaderRow \TableHead{Field} & \TableHead{Available at scoring time?} & \TableHead{Why it belongs (or does not)} \\
    \texttt{tenure\_months} & Yes & Account age recorded on the scoring date. \\
    \texttt{support\_tickets\_30d} & Yes & Recent friction; define timezone and ticket-status rules. \\
    \texttt{cancelled\_at} & No & Future outcome: using it would leak the answer. \\
    \texttt{plan} & Yes & Categorical context; preserve an \InlineCode{unknown} route. \\
    \texttt{offer\_accepted} & Depends & Include only if offer assignment preceded the outcome. \\
  \end{UpGradTable}
  \vspace{0.18cm}
  \begin{WarningBox}{Time leakage is a product bug}
    If a column is created after the decision, a high offline score can be misleading rather than useful.
  \end{WarningBox}
\end{frame}

\begin{frame}{Learning objectives}
  \begin{LearningObjectives}{By the end of this session, learners can}

\begin{frame}{A concept and its boundary}
  \begin{DefinitionBox}{Point-in-time correctness}
    Every feature value must be computed only from information available on or before the timestamp at which the prediction is made.
  \end{DefinitionBox}
  \vspace{0.28cm}
  \TwoColumnLayout{
    \begin{ExampleBox}{Correct}
      Score on 1 July using support tickets opened through 30 June.
    \end{ExampleBox}
  }{
    \begin{WarningBox}{Incorrect}
      Score on 1 July using a ``days since cancellation'' field populated on 15 July.
    \end{WarningBox}
  }
\end{frame}

\SectionDivider{02}{Make the baseline reproducible}{A small, explicit pipeline is easier to debug, explain and deploy.}

\begin{frame}[fragile]{Python: highlight the lines that matter}
  \begin{HighlightedCodeBlock}[numbers=left]{Python}{7,10,11}
features = ["monthly_spend", "tenure_months", "support_tickets", "plan"]
preprocess = ColumnTransformer([
    ("numeric", StandardScaler(), features[:3]),
    ("category", OneHotEncoder(handle_unknown="ignore"), ["plan"]),
])
model = Pipeline([
    ("prepare", preprocess),
    ("classifier", LogisticRegression(max_iter=1_000)),
])
model.fit(train[features], train["churned"])
probability = model.predict_proba(customer[features])[:, 1]
  \end{HighlightedCodeBlock}
  \vspace{0.14cm}
  \Citation{Highlighted lines are selected with \InlineCode{\{7,10,11\}}; use \InlineCode{numbers=left} when learners need to discuss a line precisely.}
\end{frame}

\begin{frame}[fragile]{SQL: create a stable feature snapshot}
  \begin{CodeBlock}[numbers=left]{SQL}
WITH feature_snapshot AS (
  SELECT
    customer_id,
    DATE '2026-06-30' AS scored_on,
    COUNTIF(created_at >= DATE_SUB(DATE '2026-06-30', INTERVAL 30 DAY))
      AS support_tickets_30d,
    MAX(plan) AS plan
  FROM analytics.support_events
  WHERE created_at < TIMESTAMP '2026-07-01 00:00:00 UTC'
  GROUP BY customer_id
)
SELECT * FROM feature_snapshot;
  \end{CodeBlock}
  \Citation{The exclusive upper bound is intentional: the snapshot must not read events that arrive after scoring begins.}
\end{frame}

\begin{frame}{A question for the room}
  \begin{QuestionBox}{Which threshold should trigger an outreach offer?}
    A probability of 0.62 is not automatically ``high enough''. What are the cost of a false positive, the value of a retained subscriber, and the daily outreach capacity?
  \end{QuestionBox}
  \vspace{0.35cm}
  \TwoColumnLayout{
    \textbf{Discuss}\begin{itemize}\item What action follows a positive prediction?\item Who is harmed by an unnecessary offer?\end{itemize}
  }{
    \textbf{Measure}\begin{itemize}\item precision among contacted subscribers\item incremental retention against a control group\end{itemize}
  }
\end{frame}

\begin{frame}{Exercise: repair the evaluation design}
  \begin{ExerciseBox}{Ten minutes, pairs}
    A team trained on January--June data and randomly split rows into train and test. They report 92\% AUC, then score July subscribers. Identify two ways this design can overstate production performance. Redesign the split in one sentence.
  \end{ExerciseBox}
  \vspace{.28cm}
  \begin{ExampleBox}{Facilitator debrief}
    A time-based split better simulates deployment. Check that the label horizon ends before the evaluation period and that one customer cannot appear on both sides through repeated snapshots.
  \end{ExampleBox}
\end{frame}

\begin{frame}{Case study: the model is right, the intervention is wrong}
  \begin{CaseStudyBox}{Retention offer fatigue}
    A model identifies high-risk annual-plan subscribers correctly. The team sends a discount to every one of them; renewal rises, but margin falls because many would have renewed without an offer.
  \end{CaseStudyBox}
  \vspace{.30cm}
  \begin{itemize}
    \item Prediction answers: \emph{who is likely to churn?}
    \item Uplift or experiment design answers: \emph{who changes behaviour because of an offer?}
    \item Operational policy answers: \emph{which action is viable and fair?}
  \end{itemize}
\end{frame}

\SectionDivider{03}{Close the loop}{A slide deck should end in a decision, a practice task and a durable mental model.}

\begin{frame}{Key takeaway}
  \begin{TakeawayBox}{Build the decision boundary before the model boundary}
    Define the decision timestamp, the intervention and the costs first. Then make features point-in-time correct, train a reproducible pipeline, and evaluate against the future workflow it will serve.
  \end{TakeawayBox}
\end{frame}

\begin{frame}{Recap}
  \begin{enumerate}
    \item A supervised-learning target is a decision framed at a timestamp.
    \item Point-in-time feature definitions prevent leakage.
    \item Pipelines keep preprocessing and model behaviour reproducible.
    \item An AWS endpoint needs lineage, monitoring and a human policy around it.
  \end{enumerate}
\end{frame}
